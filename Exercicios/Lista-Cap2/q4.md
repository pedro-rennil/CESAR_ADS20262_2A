```
int a = 1, b = 2, c = 3, d = 4; 
a += b + c;             // Valor final de a = ? 
b *= c = d + 2;         // Valores finais de b e c = ? 
d %= a + a + a;         // Valor final de d = ? 
d -= c -= b -= a;       // Valor final de d, c e b = ? 
a += b += c += 7;       // Valor final de a, b e c = ? 
```
#### Respostas

* a += b + c;
> a = a + (b + c) -> a = 1 + (2 + 3) = 6
>Estado: a = 6, b = 2, c = 3, d = 4


* b *= c = d + 2;
> d + 2 = 4 + 2 = 6
> c = 6
> b = b * c -> b = 2 * 6 = 12
>Estado: a = 6, b = 12, c = 6, d = 4

* d %= a + a + a;
> a + a + a -> (6 + 6 + 6) = 18
> d = d % 18 -> (d = 4 % 18) = 4
>Estado a = 6, b = 12, c = 6, d = 4


* d -= c -= b -= a;
> d -= a -> b = 12 - 6 = 6
> c -= b -> c = 6 - 6 = 0
> d -= c -> d = 4 - 0  = 4
> Estado: a = 6, b = 6, c = 0, d = 4

* a += b += c += 7;

> c += 7 -> c = 0 + 7 = 7
> b += c -> b = 6 + 7 = 13
> a+= b -> a = 6 + 13 = 19
>Estado: a = 19, b = 13, c = 7, d = 4
